"""Bounded paired audit: only token_metadata changes; no saved model weights."""

import gc
import importlib.util
import json
import time
from dataclasses import asdict
from pathlib import Path

import torch

import minifrontier.models.minifrontier1.modeling as modeling
from minifrontier.data import sha256
from minifrontier.models.minifrontier1 import MiniFrontier1Config, MiniFrontier1ForCausalLM
from minifrontier.models.minifrontier1.processing import token_metadata
from minifrontier.provenance import source_identity
from minifrontier.training.minifrontier1_optim import QuantileBalance, make_optimizer

ROOT = Path(__file__).resolve().parent
spec = importlib.util.spec_from_file_location("metadata_before", ROOT / "processing-before.py")
before_module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(before_module)
functions = {"before": before_module.token_metadata, "after": token_metadata}
config_path = Path("configs/minifrontier1/model_228m_native.json")
config = MiniFrontier1Config(**json.loads(config_path.read_text()))
torch.set_num_threads(2)
torch.cuda.set_per_process_memory_fraction(21 * 1024**3 / torch.cuda.get_device_properties(0).total_memory)
report = dict(
    format="mf1-packed-metadata-audit-v1",
    state="running",
    source=source_identity(),
    changed_file="minifrontier/models/minifrontier1/processing.py",
    before_sha256=sha256(ROOT / "processing-before.py"),
    after_sha256=sha256("minifrontier/models/minifrontier1/processing.py"),
    runner_sha256=sha256(__file__),
    config=asdict(config),
    config_sha256=sha256(config_path),
    device=torch.cuda.get_device_name(),
    torch=str(torch.__version__),
    cuda=torch.version.cuda,
    seed=42,
    scope="fixed synthetic packed text; bounded diagnosis, not production qualification or capability evidence",
    controls=dict(length=512, microbatch=8, global_input=16384, segments_per_row=4, warmup=3, measured=8),
    metadata={},
    optimizer_updates={},
)


def save():
    (ROOT / "report.json").write_text(json.dumps(report, indent=2, allow_nan=False) + "\n")


save()
try:
    for packed in (False, True):
        ids = torch.full((8, 512), 25, device="cuda", dtype=torch.long)
        segments = torch.arange(512, device="cuda").div(128, rounding_mode="floor").expand(8, -1) if packed else None
        outputs = [function(ids, config, segment_ids=segments) for function in functions.values()]
        for key in outputs[0]:
            torch.testing.assert_close(outputs[0][key], outputs[1][key], atol=0, rtol=0)
        entry = dict(exact_equal=True)
        for label, function in functions.items():
            for _ in range(3):
                function(ids, config, segment_ids=segments)
            torch.cuda.synchronize()
            started = time.monotonic()
            for _ in range(20):
                function(ids, config, segment_ids=segments)
            torch.cuda.synchronize()
            elapsed = time.monotonic() - started
            with torch.profiler.profile(activities=[torch.profiler.ProfilerActivity.CPU, torch.profiler.ProfilerActivity.CUDA]) as trace:
                function(ids, config, segment_ids=segments)
                torch.cuda.synchronize()
            counts = {event.key: event.count for event in trace.key_averages()}
            entry[label] = dict(seconds_per_call=elapsed / 20, calls=20, scalar_reads=counts.get("aten::_local_scalar_dense", 0), kernel_launches=counts.get("cudaLaunchKernel", 0))
        report["metadata"]["packed" if packed else "unpacked"] = entry
        save()
    del ids, segments, outputs
    for label, function in functions.items():
        gc.collect()
        torch.cuda.empty_cache()
        torch.manual_seed(42)
        modeling.token_metadata = function
        model = MiniFrontier1ForCausalLM(config).to("cuda").train()
        optimizer = make_optimizer(model)
        balance = QuantileBalance(model)
        ids = torch.randint(24, config.vocab_size, (32, 512), device="cuda")
        segments = torch.arange(512, device="cuda").div(128, rounding_mode="floor").expand(32, -1)
        labels = ids.clone()
        labels[:, ::128] = -100
        ce = int(labels[:, 1:].ne(-100).sum())
        rows = []
        for step in range(11):
            optimizer.zero_grad(set_to_none=True)
            torch.cuda.reset_peak_memory_stats()
            torch.cuda.synchronize()
            started = time.monotonic()
            for lo in range(0, 32, 8):
                with torch.autocast("cuda", dtype=torch.bfloat16), balance.capture(ids[lo:lo + 8].ne(0)):
                    result = model(ids[lo:lo + 8], labels=labels[lo:lo + 8], segment_ids=segments[lo:lo + 8], return_logits=False)
                (result.loss / 4).backward()
            norm = torch.nn.utils.clip_grad_norm_(model.parameters(), 1, error_if_nonfinite=True)
            optimizer.step()
            balance.update()
            torch.cuda.synchronize()
            seconds = time.monotonic() - started
            row = dict(step=step + 1, warmup=step < 3, input_tokens=16384, ce_tokens=ce, seconds=seconds, grad_norm=float(norm), peak_allocated_gib=torch.cuda.max_memory_allocated() / 1024**3, peak_reserved_gib=torch.cuda.max_memory_reserved() / 1024**3)
            rows.append(row)
            print(json.dumps(dict(variant=label, **row)), flush=True)
            if torch.cuda.mem_get_info()[0] < 2 * 1024**3:
                raise RuntimeError("less than 2 GiB free device memory")
        measured = rows[3:]
        report["optimizer_updates"][label] = dict(rows=rows, ce_per_second=sum(row["ce_tokens"] for row in measured) / sum(row["seconds"] for row in measured))
        save()
        del model, optimizer, balance, ids, labels, segments, result, norm
    report["ending_source"] = source_identity()
    if report["ending_source"] != report["source"]:
        raise RuntimeError("source changed during measurement")
    report["state"] = "complete"
except Exception as error:
    report.update(state="failed", error=f"{type(error).__name__}: {error}")
    raise
finally:
    modeling.token_metadata = token_metadata
    save()
